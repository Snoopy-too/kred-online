💡 Quick Start Example

Using translations in a component:

import { useTranslation } from 'react-i18next';

function MyComponent() {
  const { t } = useTranslation('common');
  
  return (
    <div>
      <h1>{t('game.title')}</h1>
      <p>{t('game.tagline')}</p>
      <button>{t('actions.play')}</button>
    </div>
  );
}
